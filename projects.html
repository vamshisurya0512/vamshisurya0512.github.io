useWebSocket.ts
export const useTransactionFeed
    = (filters: Filters) => {
  const dispatch = useAppDispatch();

  useEffect(() => {
    const ws = new WebSocket(
      `${WS_URL}/transactions`
    );
    ws.onmessage = (e) => {
      const txn = JSON.parse(e.data);
      dispatch(txnReceived(txn));
    };
    return () => ws.close();
  }, [filters]);
};
